    """
    arregla valores faltantes

    La lista de columnas eliminadas es la siguiente:

    - CATEGORIA GDD
    - SALDO GEST JUD / SALDO TOTAL

    """
    D = args[0]

    D.reset_index(inplace=True)

    cols = ['DENOMINACION', 'DESC_CATEGORIA', 'CATEGORIA GDD',
            'CLASIFICACION']

    for c in cols:
        series = []
        col_id = D.columns.get_loc(c)
        for i in D.index:
            x = D.iloc[i, col_id]
            if isinstance(x, list):
                series.append(x[0])
            if isinstance(x, str):
                series.append(x)
            else:
                series.append(x)
        D[c] = series

    yield D

[docs]def load(*args):
    """
    Carga datos a la base de datos

    El archivo destino está nombrado según el archivo de configuración

    Returns:
    None
    """
    D = args[0]

    filename = '/'.join([config['files']['setsdir'],
                         config['files']['fields_pagos_yaml']])
    filename = root_dir / filename
    with open(filename, "r") as stream:
        da = yaml.safe_load(stream)

    table_name = config['fields']['db_pagos_table']
    clobber = config['DB']['clobber_deuda']
    clobber = str2bool(clobber)

    # connect DB
    fileout = '/'.join([config['files']['datadir'],
                        config['files']['database']])
    fileout = root_dir / fileout

    conn = sqlite3.connect(fileout)
    c = conn.cursor()

    query = f"""SELECT name
                FROM sqlite_master
                WHERE type='table' AND name='{table_name}';"""
    query_results = c.execute(query).fetchall()
    df = pd.DataFrame(query_results)
    table_exists = df.shape[0] > 0

    if table_exists:
        if clobber:
            script = f'DROP TABLE IF EXISTS {table_name}'
            c.execute(script)
            conn.commit()
        else:
            conn.close()
            return

    # write table
    script = f'CREATE TABLE IF NOT EXISTS {table_name} ('
    script_head = len(script)
    for k in da.keys():
        na = da[k]['name']
        ty = da[k]['type']
        script = f'{script} {na} {ty},\n'
    script = script[:-2]
    script = f'{script});'
    c.execute(script)
    conn.commit()

    vals = []
    for k in D.index:
        X = []
        for ii in da.items():
            key = ii[0]
            ty = ii[1]['type']
            tmp = D.loc[k, [key]].values[0]
            if ty == 'INT':
                X.append(int(tmp))
            elif ty == 'TINYINT':
                X.append(int(tmp))
            elif ty == 'TEXT':
                X.append(str(tmp))
            elif ty == 'DATE':
                X.append(str(tmp))
            elif ty == 'FLOAT':
                X.append(float(tmp))
            else:
                print(ty)
        t = tuple(X)
        vals.append(t)

    # insert data
    N = len(da.keys())
    col_names = script[script_head:].split()[0::2]
    col_names = ', '.join(col_names)
    col_placeholders = ('?, ' * N)[:-2]

    insert_query = (f'INSERT INTO {table_name} ({col_names}) '
                    f'VALUES ({col_placeholders});')

    c.executemany(insert_query, vals)
    conn.commit()
    conn.close()
