[docs]def step_02(*args):
    """
    Categoriza las variables que adoptan pocos valores posibles.

    La lista de columnas categorizadas es la siguiente:

    - PROTOCOLO DE GESTION: adopta los valores
      · PROTOCOLO DIFERENCIAL
      · GENERAL
    - CYQ_TIPO PROCESO: adopta los valores:
        · CONCURSO
        · QUIEBRA
        · {NULL}
    - CYQ_DESCRIPCIÓN PROCESO: adopta los valores:
        · CONCURSO PREVENTIVO
        · GRAN CONCURSO PREVENTIVO
        · QUIEBRA PEDIDA SIMPLE/QUIEBRA PROPIA SIMPLE
        · PEQUEÑO CONCURSO PREVENTIVO
        · QUIEBRA INDIRECTA
        · {NULL}
    - CATEGORIA_CC: adopta los valores:
        · ALTA
        · MEDIA ALTA
        · BAJA
        · MEDIA BAJA
        · S/C
        · {NULL}
    - CATEGORIA_PFT: adopta los valores:
        · ALTO
        · MEDIO ALTO
        · BAJO
        · MEDIO BAJO
        · S/C
        · {NULL}
    - CATEGORIA_RF: adopta los valores:
        · ALTO
        · MEDIO ALTO
        · BAJO
        · MEDIO BAJO
        · S/C
        · {NULL}
    - CATEGORIA_CT: adopta los valores:
        · MUY BUENA
        · BUENA
        · REGULAR
        · MALA
        · S/C
        · {NULL}
    """
    D = args[0]

    cats = ['PROTOCOLO DIFERENCIAL', 'GENERAL']
    status_type = CategoricalDtype(categories=cats, ordered=False)
    D['PROTOCOLO DE GESTION'] = D['PROTOCOLO DE GESTION'].astype(status_type)

    cats = ['CONCURSO', 'QUIEBRA', '']
    status_type = CategoricalDtype(categories=cats, ordered=False)
    D['CYQ_TIPO PROCESO'] = D['CYQ_TIPO PROCESO'].astype(status_type)

    cats = ['CONCURSO PREVENTIVO', 'GRAN CONCURSO PREVENTIVO',
            'QUIEBRA PEDIDA SIMPLE/QUIEBRA PROPIA SIMPLE',
            'PEQUEÑO CONCURSO PREVENTIVO', 'QUIEBRA INDIRECTA', '']
    status_type = CategoricalDtype(categories=cats, ordered=False)

    serie = D['CYQ_DESCRIPCIÓN PROCESO']
    D['CYQ_DESCRIPCIÓN PROCESO'] = serie.astype(status_type)

    cats = ['ALTA', 'MEDIA ALTA', 'BAJA', 'MEDIA BAJA', 'S/C', '']
    status_type = CategoricalDtype(categories=cats, ordered=False)
    D['CATEGORIA_CC'] = D['CATEGORIA_CC'].astype(status_type)

    cats = ['ALTO', 'MEDIO ALTO', 'BAJO', 'MEDIO BAJO', 'S/C', '']
    status_type = CategoricalDtype(categories=cats, ordered=False)
    D['CATEGORIA_PFT'] = D['CATEGORIA_PFT'].astype(status_type)

    D['CATEGORIA_RF'] = D['CATEGORIA_RF'].astype(status_type)

    cats = ['MUY BUENA', 'BUENA', 'REGULAR', 'MALA', 'S/C', '']
    status_type = CategoricalDtype(categories=cats, ordered=False)
    D['CATEGORIA_CT'] = D['CATEGORIA_CT'].astype(status_type)

    yield D

[docs]def step_04(*args):
    """
    Verifica, corrige y uniformiza el formato de las fechas.

    Columnas afectadas:
        - CYQ_FECHA
        - FECHA_EJECUCION_PPG
        - FECHA STOCK
        - ACTUALIZACION
        - MIN_VTO_ADEUDADO
        - FECHA ACTUALIZACION

    """
    D = args[0]

    fullf = "%Y-%m-%d %H:%M:%S"

    for cname in ['CYQ_FECHA']:

        dates = []
        for s in D[cname]:
            if isinstance(s, str):
                d = datetime.datetime.strptime(s, "%d/%m/%Y")
                d = datetime.datetime.strftime(d, "%Y-%m-%d")
                dates.append(d)
            elif isinstance(s, datetime.datetime):
                d = datetime.datetime.strftime(s, "%Y-%m-%d")
                dates.append(d)
            elif isinstance(s, datetime.timedelta):
                dates.append(None)
            elif isinstance(s, pd._libs.tslibs.nattype.NaTType):
                dates.append(None)
            elif isinstance(s, pd._libs.tslibs.timestamps.Timestamp):
                dates.append(None)
            else:
                dates.append(None)

        D[cname] = dates

    for cname in ['FECHA_EJECUCION_PPG', 'FECHA STOCK',
                  'ACTUALIZACION', 'MIN_VTO_ADEUDADO',
                  'FECHA ACTUALIZACION']:

        dates = []
        for s in D[cname]:
            if isinstance(s, str):
                d = datetime.datetime.strptime(s, "%Y-%m-%d")
                dates.append(d)
            elif isinstance(s, datetime.datetime):
                dates.append(pd.NaT)
            elif isinstance(s, datetime.timedelta):
                dates.append(pd.NaT)
            elif isinstance(s, pd._libs.tslibs.nattype.NaTType):
                dates.append(pd.NaT)
            elif isinstance(s, pd._libs.tslibs.timestamps.Timestamp):
                d = datetime.datetime.strptime(str(d), fullf)
                dates.append(d)
            else:
                dates.append(pd.NaT)
        D[cname] = dates

    yield D

[docs]def load(*args):
    """
    LOAD: cargar datos a la base de datos

    El archivo destino está nombrado según el archivo de configuración

    Returns:
    None
    """
    D = args[0]

    filename = '/'.join([config['files']['setsdir'],
                         config['files']['fields_deuda_yaml']])
    filename = root_dir / filename
    with open(filename, "r") as stream:
        da = yaml.safe_load(stream)

    table_name = config['fields']['db_people_table']
    clobber = config['DB']['clobber_deuda']
    clobber = str2bool(clobber)

    # connect DB
    fileout = '/'.join([config['files']['datadir'],
                        config['files']['database']])
    fileout = root_dir / fileout

    conn = sqlite3.connect(fileout)
    c = conn.cursor()

    query = f"""SELECT name
                FROM sqlite_master
                WHERE type='table' AND name='{table_name}';"""
    query_results = c.execute(query).fetchall()
    df = pd.DataFrame(query_results)
    table_exists = df.shape[0] > 0

    if table_exists:
        if clobber:
            script = f'DROP TABLE IF EXISTS {table_name}'
            c.execute(script)
            conn.commit()
        else:
            conn.close()
            return

    script = f'CREATE TABLE IF NOT EXISTS {table_name} ('
    script_head = len(script)
    for k in da.keys():
        na = da[k]['name']
        ty = da[k]['type']
        script = f'{script} {na} {ty},\n'
    script = script[:-2]
    script = f'{script});'
    c.execute(script)
    conn.commit()

    vals = []
    for k in D.index:
        X = []
        for ii in da.items():
            key = ii[0]
            ty = ii[1]['type']
            tmp = D.loc[k, [key]].values[0]

            if isinstance(tmp, pd._libs.missing.NAType):
                X.append(None)
            elif pd.isna(tmp):
                X.append(None)
            else:
                if ty == 'INT':
                    X.append(int(tmp))
                elif ty == 'TINYINT':
                    X.append(int(tmp))
                elif ty == 'TEXT':
                    X.append(str(tmp))
                elif ty == 'DATE':
                    X.append(str(tmp))
                elif ty == 'FLOAT':
                    X.append(float(tmp))
                elif ty == 'BOOL':
                    X.append(tmp)
                else:
                    print(ty)
                    # should not print anything
        t = tuple(X)
        vals.append(t)

    # insert data
    N = len(da.keys())
    col_names = script[script_head:].split()[0::2]
    col_names = ', '.join(col_names)
    col_placeholders = ('?, ' * N)[:-2]

    insert_query = (f'INSERT INTO {table_name} ({col_names}) '
                    f'VALUES ({col_placeholders});')

    c.executemany(insert_query, vals)
    conn.commit()
    conn.close()
